📋 구글 시트 & 드라이브 자동 연동 설정 (Apps Script)

① 구글시트 열기 → 상단 메뉴 [확장 프로그램] → [Apps Script]

② 기존 코드를 지우고 아래의 최적화된 백엔드 코드를 복사하여 붙여넣은 후 저장(💾)

③ 우측 상단 [배포] → [새 배포] → 유형: [웹 앱] 선택 후,

· 다음 사용자 권한으로 앱 실행: 나(내 계정)
· 액세스 권한: 모든 사용자(Anyone)

④ 생성된 웹 앱 URL을 index.html의 APPS_SCRIPT_URL 변수에 입력하십시오.

// ══════════════════════════════════════════════════════════════
// 작전명 메모리얼 Google Apps Script 백엔드 (Code.gs)
// ══════════════════════════════════════════════════════════════
const SPREADSHEET_ID = '1nWCYrJuGPscu5LmVX7nCfmXTgaOEI9ahJmXb2v4TKSg';
const DRIVE_FOLDER_ID = '1Ixoi6vlBTXfHvm6OKNILbVSu6G9MmbAQ'; // 반별 폴더 자동 생성 루트

function doGet(e) {
  return handleRequest(e ? e.parameter : {});
}

function doPost(e) {
  var params = {};
  try {
    if (e && e.postData && e.postData.contents) {
      params = JSON.parse(e.postData.contents);
    }
  } catch (err) {
    params = (e && e.parameter) || {};
  }
  return handleRequest(params);
}

function handleRequest(p) {
  var action = p.action || '';
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var res = { success: false };

  try {
    // 1. 관리자 비밀번호 서버 검증
    if (action === 'verifyAdmin') {
      var setSh = ss.getSheetByName('설정');
      var adminPass = 'admin123';
      if (setSh) {
        var vals = setSh.getDataRange().getValues();
        for (var r = 0; r < vals.length; r++) {
          if (String(vals[r][0]).trim() === '관리자비밀번호') {
            adminPass = String(vals[r][1]).trim();
            break;
          }
        }
      }
      res = { success: (String(p.password).trim() === adminPass) };
    }

    // 2. 학급 상태 불러오기 (테마, 빙고셀, 추첨번호)
    else if (action === 'getClassState') {
      var sh = getOrCreateSheet(ss, '학급상태', ['반ID', '반이름', '테마ID', '테마이름', '선생님번호', '빙고데이터', '수정시간']);
      var rows = sh.getDataRange().getValues();
      var match = null;
      for (var i = 1; i < rows.length; i++) {
        if (rows[i][0] === p.classId) {
          match = {
            classId: rows[i][0],
            className: rows[i][1],
            themeId: rows[i][2],
            themeName: rows[i][3],
            teacherNum: rows[i][4] || null,
            cells: rows[i][5] ? JSON.parse(rows[i][5]) : []
          };
          break;
        }
      }
      res = { success: true, data: match };
    }

    // 3. 학급 상태 저장 (테마 배정 또는 빙고 업데이트)
    else if (action === 'saveClassState') {
      var sh = getOrCreateSheet(ss, '학급상태', ['반ID', '반이름', '테마ID', '테마이름', '선생님번호', '빙고데이터', '수정시간']);
      var rows = sh.getDataRange().getValues();
      var rowIndex = -1;
      for (var i = 1; i < rows.length; i++) {
        if (rows[i][0] === p.classId) {
          rowIndex = i + 1;
          break;
        }
      }
      var cellsJson = JSON.stringify(p.cells || []);
      var rowData = [p.classId, p.className, p.themeId || '', p.themeName || '', p.teacherNum || '', cellsJson, new Date()];
      if (rowIndex > 0) {
        sh.getRange(rowIndex, 1, 1, rowData.length).setValues([rowData]);
      } else {
        sh.appendRow(rowData);
      }
      res = { success: true };
    }

    // 4. 소감 등록
    else if (action === 'addFeedback') {
      var sh = getOrCreateSheet(ss, '소감기록', ['시간', '반', '이름', '소감']);
      sh.appendRow([new Date(), p.className || p.classId, p.writer, p.text]);
      res = { success: true };
    }

    // 5. 소감 목록 조회
    else if (action === 'getFeedbacks') {
      var sh = getOrCreateSheet(ss, '소감기록', ['시간', '반', '이름', '소감']);
      var rows = sh.getDataRange().getValues();
      var list = [];
      var targetClass = String(p.className || p.classId || '').trim();
      for (var i = 1; i < rows.length; i++) {
        if (!targetClass || String(rows[i][1]).trim() === targetClass) {
          list.unshift({
            time: rows[i][0],
            className: rows[i][1],
            writer: rows[i][2],
            text: rows[i][3]
          });
        }
      }
      res = { success: true, list: list };
    }

    // 6. 구글 드라이브 반별 폴더에 사진 저장
    else if (action === 'uploadPhoto') {
      var rootFolder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
      var folderName = p.className || '1반';
      var subFolders = rootFolder.getFoldersByName(folderName);
      var targetFolder = subFolders.hasNext() ? subFolders.next() : rootFolder.createFolder(folderName);

      var base64Data = p.imageBase64.replace(/^data:image\/\w+;base64,/, '');
      var decoded = Utilities.base64Decode(base64Data);
      var cellIdx = p.cellId !== undefined ? (Number(p.cellId) + 1) : '';
      var filename = folderName + '_미션' + cellIdx + '_' + (p.cellTitle || '인증') + '_' + Utilities.formatDate(new Date(), 'GMT+9', 'yyyyMMdd_HHmmss') + '.jpg';
      var blob = Utilities.newBlob(decoded, p.mimeType || 'image/jpeg', filename);
      var file = targetFolder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      var fileId = file.getId();
      var photoUrl = 'https://drive.google.com/thumbnail?id=' + fileId + '&sz=w1000';
      res = { success: true, fileId: fileId, url: photoUrl, driveUrl: file.getUrl() };
    }

    // 7. 전체 학급 현황 (관리자용)
    else if (action === 'getAllClassStates') {
      var sh = getOrCreateSheet(ss, '학급상태', ['반ID', '반이름', '테마ID', '테마이름', '선생님번호', '빙고데이터', '수정시간']);
      var rows = sh.getDataRange().getValues();
      var all = [];
      for (var i = 1; i < rows.length; i++) {
        all.push({
          id: rows[i][0],
          name: rows[i][1],
          themeId: rows[i][2],
          themeName: rows[i][3],
          teacherNum: rows[i][4] || null,
          cells: rows[i][5] ? JSON.parse(rows[i][5]) : []
        });
      }
      res = { success: true, list: all };
    }

    // 8. 전체 기록 초기화 (관리자용)
    else if (action === 'resetAllData') {
      var shState = ss.getSheetByName('학급상태');
      if (shState) {
        shState.clear();
        shState.appendRow(['반ID', '반이름', '테마ID', '테마이름', '선생님번호', '빙고데이터', '수정시간']);
      }
      var shFb = ss.getSheetByName('소감기록');
      if (shFb) {
        shFb.clear();
        shFb.appendRow(['시간', '반', '이름', '소감']);
      }
      res = { success: true };
    }
  } catch (err) {
    res = { success: false, error: err.toString() };
  }

  return ContentService.createTextOutput(JSON.stringify(res))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet(ss, name, headers) {
  var sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    if (headers && headers.length) sh.appendRow(headers);
  }
  return sh;
}
